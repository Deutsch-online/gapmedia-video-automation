 const {repo, commit, filename, revision:expectedRevision}=deploymentRecovery;
 if(repo!==process.env.CROWN_HISTORY_REPO||!process.env.CROWN_HISTORY_TOKEN)throw Error('Missing private Crown recovery configuration');
 const url='https://huggingface.co/datasets/'+repo+'/resolve/'+commit+'/'+filename;
 for(let attempt=0;attempt<4&&!recoveryData;attempt++){
  try{
   const response=await fetch(url,{headers:{Authorization:'Bearer '+process.env.CROWN_HISTORY_TOKEN},signal:AbortSignal.timeout(12000)});
   if(!response.ok)throw Error('Private recovery download failed');
   const encrypted=Buffer.from(await response.arrayBuffer());
   const decipher=crypto.createDecipheriv('aes-256-gcm',crypto.createHash('sha256').update('crown52-deploy-recovery:'+secret).digest(),encrypted.subarray(0,12));
   decipher.setAuthTag(encrypted.subarray(-16));
   const data=JSON.parse(gunzipSync(Buffer.concat([decipher.update(encrypted.subarray(12,-16)),decipher.final()])).toString('utf8'));
   if(!data.state||data.revision!==expectedRevision)throw Error('Private recovery revision mismatch');
   recoveryData=data;
  }catch{
   if(attempt===3)throw Error('Unable to restore the deployment snapshot from private storage');
   await new Promise(resolve=>setTimeout(resolve,2000));
  }
 }
